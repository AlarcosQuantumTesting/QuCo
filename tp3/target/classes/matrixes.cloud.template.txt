qreg = QuantumRegister(#QUBITS#, 'q') 
creg = ClassicalRegister(#OUTPUT_QUBITS#, 'c')
circuit = QuantumCircuit(qreg, creg)

ONE = [0, 1]
ZERO = [1, 0]
#INITIALIZE#
#HADAMARDS#
#CALCULUS#

for i in range(0, #OUTPUT_QUBITS#) :
	circuit.measure(i, #OUTPUT_QUBITS#-i-1)

print("ORIGINAL CIRCUIT:")
print(circuit)
print("**********************")

service = QiskitRuntimeService()
backend = service.least_busy(operational=True, simulator=False)
sampler = Sampler(mode=backend)


start_time = time.time()
tp = transpile(circuit, backend)
end_time = time.time()

transpilation_time = end_time - start_time
print(f"Transpilation time: {transpilation_time:.4f} seconds")

start_time = time.time()
job = sampler.run([tp])
end_time = time.time()
run_time = end_time - start_time
print(f"run() time: {run_time:.4f} seconds")
 
start_time = time.time()
result = job.result()
end_time = time.time()
result_time = end_time - start_time
print(f"result() time: {result_time:.4f} seconds")


histogram = result.get_counts()
