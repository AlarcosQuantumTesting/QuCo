def convert_histogram_to_decimal_list(histogram):
    """
    Convierte un histograma de conteos (claves binarias con valores de ocurrencias)
    en una lista donde cada valor decimal aparece tantas veces como indica el conteo,
    ordenados por sus valores decimales.

    Args:
        histogram (dict): Diccionario con claves binarias y valores de conteos.

    Returns:
        list: Lista con valores decimales repetidos según los conteos del histograma, ordenados.
    """
    decimal_list = []

    # Ordenar las claves binarias por su valor decimal
    sorted_histogram = sorted(histogram.items(), key=lambda x: int(x[0], 2))

    # Iterar sobre las claves y valores del histograma ordenado
    for binary_key, count in sorted_histogram:
        # Convertir la clave binaria a decimal
        decimal_value = int(binary_key, 2)
        # Agregar el valor decimal `count` veces a la lista
        decimal_list.extend([decimal_value] * count)

    # Guardar el resultado en un archivo
    with open('/Users/macariopolousaola/Desktop/results.txt', 'w') as file:
        file.write("\n".join(map(str, decimal_list)))

    return decimal_list
